## 4、网络协议

### 4.1、基本概念

### 4.3、TCP 和 UDP 协议

#### 4.3.1、TCP协议：传输控制协议

A、使用TCP协议前，须先建立TCP连接，形成传输数据通道。

B、传输前，==采用”三次握手“方式，能够确保数据能够接收到==，==可靠性高==。

C、TCP协议进行通信的两个应用进程：客户端、服务端。

D、在连接中可进行大数据量的传输。

E、==传输完毕，需释放已建立的连接，效率低==。



#### 4.3.2、UDP协议：用户数据协议

A、将数据、源、目的封装成数据包，不需要建立连接。

B、==每个数据报的大小限制在64K内，不适合传输大量数据==。

C、因无需连接，故是不可靠的。

​	==UDP只负责将数据打包发送出去，不会进行可靠性验证，因此是不可靠的==。

D、发送数据结束时无需释放资源（因为不是面向连接的），速度快。

E、举例：厕所通知：发短信。



## 5、==InetAddress类==

### 5.1、相关方法

A、getLocalHost：获取本地主机的 IP 地址和主机名

B、getByName：根据主机名或 IP 地址字符串，获取对应的 `InetAddress` 对象

```java
package com.cyl.api;

import java.net.InetAddress;
import java.net.UnknownHostException;

/**
 * @Author：cyl
 * @CreateTime：2025-06-25
 * @Description：演示InetAddress的使用
 * @version：1.0
 */
public class Api {
    public static void main(String[] args) throws UnknownHostException {
        //1、获取本机的InetAddress对象
        InetAddress localHost = InetAddress.getLocalHost();
        //会输出当前主机的：主机名 和 IP地址
        System.out.println(localHost); //Administrator/192.168.2.100

        //2、根据指定的主机名,获取InetAddress对象
        InetAddress host1 = InetAddress.getByName("Administrator");
        System.out.println(host1); //Administrator/192.168.2.100

        //3、根据指定的域名,获取InetAddress对象，如：www.baidu.com
        InetAddress host2 = InetAddress.getByName("www.baidu.com");
        System.out.println(host2); //www.baidu.com/183.2.172.177

        //4、通过InetAddress对象，获取：主机IP地址
        String hostAddress = host2.getHostAddress();
        System.out.println(hostAddress); //183.2.172.177

        //5、通过InerAddress对象，获取：主机名 或 域名
        String hostName = host2.getHostName();
        System.out.println(hostName); //www.baidu.com
    }
}
```

```java
import java.net.Socket;

/**
 * @Author：cyl
 * @CreateTime：2025-06-25
 * @Description：服务端
 * @version：1.0

public class SocketService {
    public static void main(String[] args) throws IOException {
        //思路分析
        //1、在本机的9999端口监听，等待连接
        //细节：设置端口号时，该端口号必须是空闲的，没有被其他服务使用，否则会抛异常
        //在没有任务连接端口服务时，会阻塞在这里等待任务连接
        ServerSocket serverSocket = new ServerSocket(9999);
        System.out.println("服务端在9999端口监听，等待连接"); //没有任务连接端口，程序会阻塞在这里等待连接

        //2、如果没有客户端连接端口时，会进入阻塞状态，等待客户端连接
        //    如果有客户端连接端口时，会返回一个Socket对象
        //    serverSocket可以通过accept()来返回多个socket对象，用于多个客户端连接服务器
        Socket socket = serverSocket.accept();
        System.out.println("服务端获取到socket对象：" + socket.getClass());

        //3、通过socket.getInputStream()读取用户端传入数据通道的数据，并显示
        InputStream inputStream = socket.getInputStream();
        byte[] buf = new byte[1024];
        int readNum = 0;
        while ((readNum = inputStream.read(buf)) != -1) {
            System.out.println(new String(buf, 0, readNum)); //根据实际读取到的实际数据个数，创建字符串输出信息
        }

        //4、关闭流、socket、serverSocket
        inputStream.close();
        socket.close();
        serverSocket.close();
    }
}
```

### 6.3、UDP编程

#### 6.3.1、基本介绍

A、类 ==DatagramSocket== 和 ==DatagramPacket== 实现了基于UDP协议的网络程序。

B、UDP数据报通过数据报套接字 DatagramSocket 发送和接收数据，==系统不保证UDP数据报一定能够安全送到目的地，也不能确定什么时候可以抵达==。

C、==DatagramPacket对象，封装了UDP数据报，在数据报中包含了：发送端的 IP地址和端口号 以及 接收端的 IP地址和端口号==。

D、UDP协议中每个数据报都给出了完整的地址信息，因此无须建立发送方和接收方的连接，就能通信。

```java
package com.udp_;

import java.io.IOException;
import java.net.DatagramPacket;
import java.net.DatagramSocket;
import java.net.InetAddress;
import java.net.SocketException;

/**
 * @Author：cyl
 * @CreateTime：2025-07-09
 * @Description：接收端A
 * @version：1.0
 */
public class UDPReceiverA {
    public static void main(String[] args) throws IOException {
        //1、接收端在本机IP的9999端口监听，准备接收发送端的数据
        DatagramSocket datagramSocket = new DatagramSocket(9999);
        //2、创建一个DatagramPacket来接收数据
        byte[] bytes = new byte[1024];
        DatagramPacket datagramPacket = new DatagramPacket(bytes, bytes.length);
        //3、调用receive方法来接收发送端传来的数据
        datagramSocket.receive(datagramPacket);
        //4、将接收到的datagramPacket数据包进行拆包显示
        byte[] data = datagramPacket.getData(); //接收到的数据内容
        int length = datagramPacket.getLength(); //接收到的数据长度
        String message = new String(data, 0, length); //将数据拆包得到的字节数组封装为字符串输出
        System.out.println(message);

        byte[] backMessage = "好的，明天见".getBytes();
        datagramPacket = new DatagramPacket(backMessage, backMessage.length, InetAddress.getByName("192.168.2.100"), 8888);
        datagramSocket.send(datagramPacket);

        datagramSocket.close();
    }
}
```

```java
package com.homework;

import java.io.*;
import java.net.ServerSocket;
import java.net.Socket;

public class HomeWork03Server {
    public static void main(String[] args) throws IOException {
        ServerSocket serverSocket = new ServerSocket(8888);
        System.out.println("服务器启动，等待客户端连接...");
        Socket socket = serverSocket.accept();
        System.out.println("客户端连接成功...");

        //读取客户端要下载的音乐文件名（只读取一行 ---> 以\n为结束标记）
        BufferedReader br = new BufferedReader(new InputStreamReader(socket.getInputStream()));
        //接收音乐名
        String musicName = br.readLine(); // 客户端发送端数据必须以\n换行结束，客户端写入也要 flush
        System.out.println("收到客户端请求音乐名: " + musicName);

        //判断音乐文件是否存在
        //a、如果找到了对应的音乐文件，则返回该名称所对应的音乐文件
        //b、如果没有找到所对应的音乐文件，则返回稻香.mp3音乐文件
        String filePath = "src/" + musicName + ".mp3";
        File musicFile = new File(filePath);
        if (musicFile.exists()) {
            System.out.println("成功找到音乐文件" + musicName);
            filePath = "src/" + musicName  + ".mp3";
        } else {
            System.out.println("文件不存在，发送默认音乐");
            filePath = "src/稻香.mp3";
        }

        //获取音乐文件名，传输给客户端，用于命名拷贝后的新音乐文件
        BufferedWriter bw = new BufferedWriter(new OutputStreamWriter(socket.getOutputStream()));
        File file = new File(filePath);
        //getName()返回的是构造File对象时传入路径中最后的那一部分
        //filePath路径下文件包含扩展名，因此getName()返回的内容为：音乐名称.扩展名
        bw.write(file.getName()); 
        bw.newLine(); //换行符，结束标记
        bw.flush();

        //向客户端发送音乐文件
        BufferedOutputStream bos = new BufferedOutputStream(socket.getOutputStream());
        //读取文件并发送
        BufferedInputStream fileIn = new BufferedInputStream(new FileInputStream(filePath));
        byte[] buffer = new byte[1024];
        int len;
        while ((len = fileIn.read(buffer)) != -1) {
            bos.write(buffer, 0, len);
        }
        bos.flush(); // flush 确保全部发送
        socket.shutdownOutput(); // 通知客户端“发完了”

        fileIn.close();
        bos.close();
        bw.close();
        br.close();
        socket.close();
        serverSocket.close();
    }
}
```

```java
package com.homework;

import java.io.*;
import java.net.InetAddress;
import java.net.Socket;
import java.util.Scanner;

public class HomeWork03Client {
    public static void main(String[] args) throws IOException {
        Socket socket = new Socket(InetAddress.getByName("192.168.2.100"), 8888);

        BufferedWriter bw = new BufferedWriter(new OutputStreamWriter(socket.getOutputStream()));

        Scanner scanner = new Scanner(System.in);
        System.out.println("已连接服务器");
        // 1. 输入请求服务器端下载的音乐名
        System.out.print("请输入要下载的音乐名称（不含扩展名）：");
        String musicName = scanner.nextLine();

        // 2. 向服务器端发送要下载的音乐名(不包含扩展名)
        bw.write(musicName);
        bw.newLine(); // 必须加换行，让服务端 readLine() 不阻塞
        bw.flush();
        socket.shutdownOutput();
        System.out.println("已发送音乐名称，准备接收音乐文件...");

        //3.1、接收服务器端发送过来的音乐名
        BufferedReader br = new BufferedReader(new InputStreamReader(socket.getInputStream()));
        String backMusicName = br.readLine();
        System.out.println("服务器发送过来的音乐文件名为：" + backMusicName);

        // 3. 接收服务器端发送来的音乐文件内容，并将音乐文件写入到磁盘中
        BufferedInputStream bis = new BufferedInputStream(socket.getInputStream());
        BufferedOutputStream bos = new BufferedOutputStream(new FileOutputStream("e:/" + backMusicName));
        byte[] buffer = new byte[1024];
        int len;
        while ((len = bis.read(buffer)) != -1) {
            bos.write(buffer, 0, len);
        }
        bos.flush();

        bos.close();
        bis.close();
        br.close();
        bw.close();
        socket.close();
    }
```
